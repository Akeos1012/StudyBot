 185    def is_structural(self, line: str, context: str = "") -> Tuple[bool, str]:
 186        if not line:
 187            return False, ""
 188
 189        stripped = line.strip()
 190        if not stripped.startswith("#"):
 191            return False, ""
 192
 193        heading = re.sub(r"^#+\s*", "", stripped).strip()
 194        if not heading:
 195            return False, ""
 196
 197        heading_lower = heading.lower()
 198
 199        for pred in HeadingFilter.CONCEPT_PREDICATES:
 200            if pred in heading_lower:
 201                return False, ""
 202
 203        words = heading_lower.split()
 204        if words:
 205            last_word = words[-1].rstrip(".,;:")
 206            if last_word in HeadingFilter.ORGANIZATIONAL_SUFFIXES:
 207                return True, f"organizational_suffix: {last_word}"
 208
 209        if len(words) <= 4 and not any(
 210            p in heading_lower for p in HeadingFilter.CONCEPT_PREDICATES
 211        ):
 212            return True, "title_only_heading"
 213
 214        structural_markers = {
 215            "overview",
 216            "summary",
 217            "introduction",
 218            "conclusion",
 219            "analysis",
 220        }
 221        for marker in structural_markers:
 222            if marker in heading_lower:
 223                return True, f"structural_marker: {marker}"
 224
 225        return False, ""

 484    def extract(self, text: str) -> Optional[str]:
 485        """Extract a concept from text using strict semantic rules."""
 486        if not text:
 487            return None
 488
 489        text = text.strip()
 490
 491        # Pattern 1: "X is Y" - subject is X (strongest signal)
 492        match = re.search(
 493            r"^([A-Z][a-zA-Z\s]{2,})\s+(is|are|refers to|means|stands for)",
 494            text,
 495            re.IGNORECASE,
 496        )
 497        if match:
 498            concept = match.group(1).strip()
 499            concept = " ".join(concept.split()[:4])
 500            if self._is_canonical_concept(concept):
 501                return self._normalize_concept(concept)
 502
 503        # Pattern 2: "X allows/stores..." - subject is X
 504        match = re.search(
 505            r"^([A-Z][a-zA-Z\s]{2,})\s+(allows|enables|provides|stores|manages|"
 506            r"reduces|improves|uses|supports|offers|helps|contains|includes)",
 507            text,
 508            re.IGNORECASE,
 509        )
 510        if match:
 511            concept = match.group(1).strip()
 512            concept = " ".join(concept.split()[:4])
 513            if self._is_canonical_concept(concept):
 514                return self._normalize_concept(concept)
 515
 516        # Pattern 3: "called X" or "known as X"
 517        match = re.search(
 518            r"(?:called|known as|termed)\s+([A-Z][a-zA-Z\s]{2,})(?:\.|$)",
 519            text,
 520            re.IGNORECASE,
 521        )
 522        if match:
 523            concept = match.group(1).strip()
 524            concept = " ".join(concept.split()[:4])
 525            if self._is_canonical_concept(concept):
 526                return self._normalize_concept(concept)
 527
 528        # Pattern 4: "X - Y" or "X — Y" (subject is X)
 529        match = re.search(r"^([A-Z][a-zA-Z\s]{2,})\s*[—\-]\s+", text)
 530        if match:
 531            concept = match.group(1).strip()
 532            concept = " ".join(concept.split()[:4])
 533            if self._is_canonical_concept(concept):
 534                return self._normalize_concept(concept)
 535
 536        # Pattern 5: "X (Y)" - X is the concept, Y is the acronym
 537        match = re.search(r"^([A-Z][a-zA-Z\s]{2,})\s*\(([A-Z]{2,})\)", text)
 538        if match:
 539            concept = match.group(1).strip()
 540            concept = " ".join(concept.split()[:4])
 541            if self._is_canonical_concept(concept):
 542                return self._normalize_concept(concept)
 543
 544        # Pattern 6: "X:" - X is the concept
 545        match = re.search(r"^([A-Z][a-zA-Z\s]{2,})\s*:", text)
 546        if match:
 547            concept = match.group(1).strip()
 548            concept = " ".join(concept.split()[:4])
 549            if self._is_canonical_concept(concept):
 550                return self._normalize_concept(concept)
 551
 552        return None

 638    def _is_canonical_concept(self, concept: str) -> bool:
 639        """
 640        Determine if a concept is a canonical noun phrase suitable for quiz questions.
 641
 642        Returns True only if ALL of:
 643        - Is a noun or noun phrase
 644        - Names a thing, system, algorithm, protocol, hardware component,
 645          data structure, language, model, metric, process, architecture, or technology
 646        - Could realistically be the answer to a quiz question
 647        """
 648        if not concept:
 649            return False
 650
 651        # Validate against schema first
 652        is_valid, _ = self.concept_validator.is_valid(concept)
 653        if not is_valid:
 654            return False
 655
 656        concept_lower = concept.lower()
 657        words = concept_lower.split()
 658        if not self._has_valid_concept_structure(concept):
 659            logger.debug(f"Rejected '{concept}': invalid concept structure")
 660            return False
 661
 662        if not words:
 663            return False
 664
 665        first_word = words[0]
 666        last_word = words[-1]
 667
 668        # 1. REJECT filler starts
 669        if first_word in self.FILLER_STARTS:
 670            logger.debug(f"Rejected '{concept}': starts with filler '{first_word}'")
 671            return False
 672
 673        # 2. REJECT verb starts
 674        if first_word in self.VERB_STARTS:
 675            logger.debug(f"Rejected '{concept}': starts with verb '{first_word}'")
 676            return False
 677
 678        # 3. REJECT grammatical predicates (contains verbs)
 679        for verb in self.PREDICATE_VERBS:
 680            # Check if the verb appears as a whole word in the concept
 681            if re.search(r"\b" + verb + r"\b", concept_lower):
 682                logger.debug(f"Rejected '{concept}': contains predicate verb '{verb}'")
 683                return False
 684
 685        # 4. REJECT section labels at end (unless overridden)
 686        if last_word in self.SECTION_LABELS:
 687            if len(words) == 1:
 688                return True
 689
 690            logger.debug(
 691                f"Rejected '{concept}': ends with section label '{last_word}'"
 692            )
 693            return False
 694
 695        # 5. Check for section labels in the middle
 696        for word in words:
 697            if word in self.SECTION_LABELS and len(words) > 1:
 698                if concept not in self._concept_type_overrides:
 699                    logger.debug(
 700                        f"Rejected '{concept}': contains section label '{word}'"
 701                    )
 702                    return False
 703
 704        # 6. REJECT verb phrase patterns (e.g., "Focuses On How", "Improves Time")
 705        verb_phrase_patterns = [
 706            r"^focuses\s+on\s+",
 707            r"^focusing\s+on\s+",
 708            r"^improves\s+",
 709            r"^improving\s+",
 710            r"^allows\s+the\s+",
 711            r"^provides\s+",
 712            r"^providing\s+",
 713            r"^supports\s+",
 714            r"^supporting\s+",
 715            r"^enables\s+",
 716            r"^enabling\s+",
 717            r"^uses\s+",
 718            r"^using\s+",
 719            r"^contains\s+",
 720            r"^containing\s+",
 721            r"^includes\s+",
 722            r"^including\s+",
 723            r"^manages\s+",
 724            r"^managing\s+",
 725            r"^reduces\s+",
 726            r"^reducing\s+",
 727            r"^creates\s+",
 728            r"^creating\s+",
 729            r"^processes\s+",
 730            r"^processing\s+",
 731            r"^handles\s+",
 732            r"^handling\s+",
 733            r"^generates\s+",
 734            r"^generating\s+",
 735            r"^runs\s+",
 736            r"^running\s+",
 737        ]
 738        for pattern in verb_phrase_patterns:
 739            if re.match(pattern, concept_lower):
 740                logger.debug(f"Rejected '{concept}': matches verb phrase pattern")
 741                return False
 742
 743        # 7. REJECT question-word starts
 744        if first_word in {
 745            "where",
 746            "when",
 747            "why",
 748            "how",
 749            "what",
 750            "which",
 751            "who",
 752            "whom",
 753        }:
 754            logger.debug(
 755                f"Rejected '{concept}': starts with question word '{first_word}'"
 756            )
 757            return False
 758
 759        # 8. REJECT phrases with "The" at start
 760        if concept_lower.startswith("the "):
 761            logger.debug(f"Rejected '{concept}': starts with 'The'")
 762            return False
 763
 764        # 9. REJECT concepts that are clearly sentences or fragments
 765        # Check for ending punctuation
 766        if concept.endswith(".") or concept.endswith("!") or concept.endswith("?"):
 767            return False
 768
 769        # Check for multiple sentences (periods inside)
 770        if concept.count(".") > 1:
 771            return False
 772
 773        # 10. REJECT concepts that are too vague
 774        vague_terms = {
 775            "system",
 776            "process",
 777            "method",
 778            "approach",
 779            "technique",
 780            "mechanism",
 781            "framework",
 782        }
 783        if len(words) == 1:
 784            if first_word in vague_terms:
 785                logger.debug(f"Rejected '{concept}': too vague")
 786                return False
 787
 788        # 11. Check for lowercase words in the middle (should be all capitalized for multi-word)
 789        if len(words) >= 2:
 790            for w in words:
 791                # Allow acronyms
 792                if w.upper() in self._valid_acronyms:
 793                    continue
 794                # Check if the word is capitalized or is a common word like "and", "of", "for"
 795                if not w[0].isupper() and w not in {
 796                    "and",
 797                    "of",
 798                    "for",
 799                    "in",
 800                    "on",
 801                    "at",
 802                    "to",
 803                    "by",
 804                    "with",
 805                    "without",
 806                }:
 807                    # It might still be valid if it's a proper noun phrase
 808                    pass
 809
 810        return True
